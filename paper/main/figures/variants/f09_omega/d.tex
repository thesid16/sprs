% f09_omega variant d -- distribution of cycles/Omega over the 39 HW-validated
%   instances, in bins of width 0.5.
% Data: figures/omega_by_instance.tex (from scripts/gen_figures.py over
%   /home/rohit/sprs-fix/data/results/live_hw_results_p1_maxpe.csv, max-over-PEs).
%   Bin counts, left edges 1.0 upward in steps of 0.5:
%   4, 8, 8, 0, 6, 3, 3, 3, 0, 0, 0, 1, 1, 0, 1, 0, 1  (sums to 39).
%   Minimum \SlackBest{} (macro \SlackBest); median 2.34 over the same 39 ratios.
% Strategy: argue the DISTRIBUTION rather than the per-instance detail. Variants
%   a, b and c all keep one mark per instance; here the reader is shown that the
%   mass sits between 1.5x and 3.5x with a thin right tail, which is the shape
%   the "additive, not multiplicative" reading in the text depends on.
% Target: IEEE single column, 3.5in = 8.9cm. Drawn at 8.6cm.
\begin{tikzpicture}[
  bin/.style={draw=spBlue, fill=spBlue!26, line width=0.4pt},
]
\begin{axis}[
  width=8.6cm, height=5.4cm,
  xlabel={champion cycles $/\ \Omega$}, ylabel={instances},
  xmin=1.0, xmax=9.5, ymin=0, ymax=9.9,
  xtick={1,2,3,4,5,6,7,8,9},
  ytick={0,2,4,6,8},
  tick label style={font=\scriptsize, text=spInk},
  label style={font=\scriptsize, text=spInk},
  grid=major, major grid style={spGrid, very thin},
  clip=false,
]
% Bars drawn explicitly (rather than with ybar interval) so the x ticks stay on
% the true ratio scale instead of being moved to bin boundaries. One line per
% non-empty bin; the empty bins (2.5--3.0, 5.0--6.5, 7.5--8.0, 8.5--9.0) draw
% nothing, which is what makes the gap in the tail visible.
\addplot[draw=none, forget plot] coordinates {(1.0,0) (9.5,0)};

\filldraw[bin] (axis cs:6.5,0) rectangle (axis cs:7,1);
\filldraw[bin] (axis cs:7,0) rectangle (axis cs:7.5,1);
\filldraw[bin] (axis cs:8,0) rectangle (axis cs:8.5,1);
\filldraw[bin] (axis cs:9,0) rectangle (axis cs:9.5,1);

  {$n=39$ HW-validated instances\\bins of width $0.5$\\
   no instance reaches $\Omega$};
\end{axis}
\end{tikzpicture}
